/*
 * Warehouse Operator - Custom Item Definitions
 *
 * WHO_PumpKey: quest token for WHO_Q001 "Prime the Pump" - recover from the gas
 * station, deliver to the extraction crate.
 *
 * Templated from the vanilla B42 key (media/scripts/generated/items/key.txt ->
 * item CarKey): item name changed, DisplayName added (vanilla keys use translation
 * files we don't ship), and two inapplicable tags dropped - base:carkey (wrong for
 * a pump key, could touch vehicle systems) and base:morewhennozombies (a loot-
 * rarity modifier, moot since we hand-spawn the key via the lazy-spawn queue).
 *
 * WHY: the previous hand-written minimal definition used the legacy `Type = Normal`.
 * It PARSED and registered (getScriptManager():getItem returned it), but B42's
 * InventoryItemFactory builds the InventoryItem from the new `ItemType = base:<type>`
 * system. Without `ItemType`, the factory produced a null item -> AddItem returned
 * nil and AddWorldInventoryItem threw NullPointerException at createItemInternal
 * (setAlcoholPower). Using the full vanilla template gives PZ everything it needs.
 */
module Base
{
    item WHO_PumpKey
    {
        DisplayCategory  = Security,
        Weight           = 0.05,
        ItemType         = base:key,
        DisplayName      = Fuel Pump Key,
        Icon             = CarKey,
        MetalValue       = 7.0,
        WorldStaticModel = CarKeys,
        Tags             = base:fitswallet,
        OriginX          = 0,
        OriginY          = 0,
        originZ          = 0,
    }
}
